Select the fitted bank whose plan digest matches those exact plan bytes. Choose a new output directory outside the checkout.

```sh
uv sync --locked --python 3.11.14
uv run --no-sync empirical-lawhood --project-root "$PWD" campaign matrix-response-author \
  --config experiments/causal-response/causal-author.json \
  --source-root "$PREPARED_RESPONSE_HELD_SOURCE_ROOT" \
  --plan "$CAUSAL_RESPONSE_PLAN" \
  --design-packet "$CAUSAL_RESPONSE_DESIGN_PACKET" \
  --prior-exposure "$CAUSAL_RESPONSE_PRIOR_EXPOSURE" \
  --model-bank "$CAUSAL_RESPONSE_MODEL_BANK" \
  --output-dir "$CAUSAL_RESPONSE_NEW_AUTHORING_DIRECTORY"
```
